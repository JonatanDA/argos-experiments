function Print_tracker_python(obs_ang_offset_global, x_mark, y_mark, yellow_blobs, green_blobs)
    local py_str = "[python];observation;" .. tostring(obs_ang_offset_global) .. ";" .. tostring(x_mark) .. ";" .. tostring(y_mark) .. ";"
    
    local yb_strs = {}
    if yellow_blobs then
        for _, b in ipairs(yellow_blobs) do
            table.insert(yb_strs, b.angle .. "," .. b.distance .. "," .. (b.type or 2))
        end
    end
    py_str = py_str .. table.concat(yb_strs, "|") .. ";"
    
    local gb_strs = {}
    if green_blobs then
        for _, b in ipairs(green_blobs) do
            table.insert(gb_strs, b.angle .. "," .. b.distance)
        end
    end
    py_str = py_str .. table.concat(gb_strs, "|")
    
    print(py_str)
end
